<script>
                                                                                          // Constants and DOM Elements
                                                                                          const cardsContainer = document.getElementById('cards-container');
                                                                                          const prevButton = document.getElementById('prev-button');
                                                                                          const nextButton = document.getElementById('next-button');
                                                                                          const prevButtonMobile = document.getElementById('prev-button-mobile');
                                                                                          const nextButtonMobile = document.getElementById('next-button-mobile');
                                                                                          const reviewCards = document.querySelectorAll('.review-card');
                                                                                          const totalCards = reviewCards.length;
                                                                                          const GAP = 16; // Based on Tailwind's 'gap-4' (4 * 4px = 16px)
                                                                                        
                                                                                          let currentIndex = 0;
                                                                                          let cardsPerView = 4; // Default for large screens
                                                                                        
                                                                                          /**
                                                                                           * Determines how many cards should be visible based on screen width.
                                                                                           * Corresponds to Tailwind breakpoints.
                                                                                           * @returns {number} The number of cards to display.
                                                                                           */
                                                                                          function getCardsPerView() {
                                                                                            const width = window.innerWidth;
                                                                                            if (width < 768) {
                                                                                              return 1; // Mobile (0px to 767px)
                                                                                            } else if (width < 1024) {
                                                                                              return 2; // Tablet (768px to 1023px)
                                                                                            } else if (width < 1280) {
                                                                                              return 3; // Small Desktop (1024px to 1279px)
                                                                                            } else if (width < 1440) {
                                                                                              return 3; // Medium Desktop (1280px to 1439px)
                                                                                            } else {
                                                                                              return 4; // Large Desktop (1440px and up)
                                                                                            }
                                                                                          }
                                                                                        
                                                                                          /**
                                                                                           * Calculates and applies the required width and transform for the slider elements.
                                                                                           */
                                                                                          function updateSlider() {
                                                                                            cardsPerView = getCardsPerView();
                                                                                        
                                                                                            // 1. Update Card Widths
                                                                                            // Card width = (Container Width - Total Gaps) / cardsPerView
                                                                                            // The container width is effectively 100% of its parent.
                                                                                            const gapCount = cardsPerView - 1;
                                                                                            const cardWidth = `calc((100% - ${gapCount * GAP}px) / ${cardsPerView})`;
                                                                                        
                                                                                            reviewCards.forEach(card => {
                                                                                              card.style.width = cardWidth;
                                                                                            });
                                                                                        
                                                                                            // 2. Calculate the maximum number of moves possible
                                                                                            // The last movable index is when the first card in the view is at position: totalCards - cardsPerView
                                                                                            const maxIndex = totalCards - cardsPerView;
                                                                                        
                                                                                            // Ensure currentIndex does not exceed maxIndex
                                                                                            if (currentIndex > maxIndex || currentIndex < 0) {
                                                                                              currentIndex = Math.max(0, Math.min(currentIndex, maxIndex));
                                                                                            }
                                                                                        
                                                                                            // 3. Calculate Translation
                                                                                            // Translation is based on the starting position of the card at currentIndex.
                                                                                            // Distance = (Card Width + Gap) * currentIndex
                                                                                            const translateDistance = currentIndex * (cardsContainer.children[0].offsetWidth + GAP);
                                                                                            cardsContainer.style.transform = `translateX(-${translateDistance}px)`;
                                                                                        
                                                                                            // 4. Update Button States
                                                                                            prevButton.disabled = prevButtonMobile.disabled = currentIndex === 0;
                                                                                            nextButton.disabled = nextButtonMobile.disabled = currentIndex >= maxIndex;
                                                                                          }
                                                                                        
                                                                                          /**
                                                                                           * Moves the slider to the previous set of cards.
                                                                                           */
                                                                                          function slidePrev() {
                                                                                            // Move back by the number of cards currently visible
                                                                                            currentIndex = Math.max(0, currentIndex - cardsPerView);
                                                                                            updateSlider();
                                                                                          }
                                                                                        
                                                                                          /**
                                                                                           * Moves the slider to the next set of cards.
                                                                                           */
                                                                                          function slideNext() {
                                                                                            // Move forward by the number of cards currently visible
                                                                                            const maxIndex = totalCards - cardsPerView;
                                                                                            currentIndex = Math.min(maxIndex, currentIndex + cardsPerView);
                                                                                            updateSlider();
                                                                                          }
                                                                                        
                                                                                        
                                                                                          // --- Event Listeners ---
                                                                                        
                                                                                          // Bind slide functions to the navigation buttons (both desktop and mobile controls)
                                                                                          prevButton.addEventListener('click', slidePrev);
                                                                                          nextButton.addEventListener('click', slideNext);
                                                                                          prevButtonMobile.addEventListener('click', slidePrev);
                                                                                          nextButtonMobile.addEventListener('click', slideNext);
                                                                                        
                                                                                          // Update the slider on initial load and whenever the window resizes
                                                                                          window.addEventListener('load', updateSlider);
                                                                                          window.addEventListener('resize', updateSlider);
                                                                                        
                                                                                          // Initial call to set up the slider correctly on load
                                                                                          updateSlider();
</script>

                                                                                        document.addEventListener('DOMContentLoaded', function () { 
                                                                                        const track = document.getElementById('cards-container'); 
                                                                                        const prevButton = document.getElementById('prev-button'); 
                                                                                        const nextButton = document.getElementById('next-button'); 
                                                                                        const prevButtonMobile = document.getElementById('prev-button-mobile'); 
                                                                                        const nextButtonMobile = document.getElementById('next-button-mobile'); 
                                                                                        const cardWidth = 316; // 300px (min-w) + 16px (gap) 
                                                                                        let position = 0; 
                                                                                        function updateButtons() { 
                                                                                        prevButton.disabled = position >= 0; 
                                                                                        prevButtonMobile.disabled = position >= 0; 
                                                                                        // Check if we've reached the end 
                                                                                        const maxScroll = track.scrollWidth - track.parentElement.offsetWidth; 
                                                                                        nextButton.disabled = Math.abs(position) >= maxScroll; 
                                                                                        nextButtonMobile.disabled = Math.abs(position) >= maxScroll; 
                                                                                        } 
                                                                                        function slide(direction) { 
                                                                                        const containerWidth = track.parentElement.offsetWidth; 
                                                                                        const maxScroll = track.scrollWidth - containerWidth; 
                                                                                        if (direction === 'next' && Math.abs(position) < maxScroll) { 
                                                                                        position -= cardWidth; 
                                                                                        if (Math.abs(position) > maxScroll) { 
                                                                                        position = -maxScroll; 
                                                                                        } 
                                                                                        } else if (direction === 'prev' && position < 0) { 
                                                                                        position += cardWidth; 
                                                                                        if (position > 0) { 
                                                                                        position = 0; 
                                                                                        } 
                                                                                        } 
                                                                                        track.style.transform = `translateX(${position}px)`; 
                                                                                        updateButtons(); 
                                                                                        } 
                                                                                        prevButton.addEventListener('click', () => slide('prev')); 
                                                                                        nextButton.addEventListener('click', () => slide('next')); 
                                                                                        prevButtonMobile.addEventListener('click', () => slide('prev')); 
                                                                                        nextButtonMobile.addEventListener('click', () => slide('next')); 
                                                                                        // Initialize button states 
                                                                                        updateButtons();
                                                                                        });
                                                                                        </script>
